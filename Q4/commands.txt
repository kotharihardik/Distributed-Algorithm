----------------------------------------------------------
 STEP 5 - ASK SLURM FOR 8 REAL CORES
----------------------------------------------------------

salloc --nodes=1 --ntasks=8 --hint=nomultithread --time=01:00:00 --partition=debug
    # reserve cores on a compute node so my job can run
    #
    # salloc                = reserve resources and wait, interactive
    # --nodes=1             = use ONE machine.
    #                         MUST be 1 - multi-node MPI is broken here
    # --ntasks=8            = reserve 8 cores (for P=8, the biggest test)
    # --hint=nomultithread  = give me 8 REAL cores, not 8 fake hyperthreads.
    #                         MUST have this, or P=8 runs at half speed
    # --time=01:00:00       = reserve for 1 hour, then auto-release
    # --partition=debug     = which queue. "debug" is the only one here
    #
    # NOTE: my shell stays on the LOGIN node. Only mpirun sends work to
    # the compute node. So always use home-folder paths, never /tmp,
    # because /tmp is different on each machine.


----------------------------------------------------------
 STEP 6 - SET THE LAUNCHER
----------------------------------------------------------

export LAUNCH="mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np"
    # save the mpirun command in a variable, so the scripts below use it
    #
    # export        = make this variable visible to the scripts I run
    # mpirun        = the command that starts N copies of an MPI program
    # --mca pml ^ucx and --mca osc ^ucx
    #               = switch OFF the "ucx" network module.
    #                 The cluster advertises it but the library file is
    #                 missing, so without this every process prints an error.
    #                 The "^" means "not this one".
    # --bind-to core= pin each process to its own core and keep it there,
    #                 so the OS doesn't shuffle them around
    # -np           = "number of processes", the number comes right after.
    #                 The scripts add it, e.g. -np 4

-- run the example from the question paper --

mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np 4 bin/tri_mpi data/sample.txt
    # -np 4              = start 4 processes
    # bin/tri_mpi        = the program
    # data/sample.txt    = the graph file
    #
    # the answer is just one number, printed on screen:
    #     2
    # (the two triangles are {0,1,2} and {0,2,3})


-- check the MPI answer equals the sequential answer --

bin/tri_seq data/sample.txt
    # the slow 1-core version. prints the same single number.

mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np 8 bin/tri_mpi data/sample.txt
    # the 8-process version. both numbers must match.


-- a graph where I know the answer for sure --

mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np 8 bin/tri_mpi data/tri_k5.txt
    # K5 = 5 vertices, all connected to each other.
    # every group of 3 vertices is a triangle, so the answer must be
    # exactly C(5,3) = 10. Good sanity check.


-- make one graph file myself --

mkdir -p data/generated

bin/gen_graph 6000 750000 12345 data/generated/Large.txt
    # arguments in order:  V  E  seed  outfile
    #   V 6000    = number of vertices (0 to 5999)
    #   E 750000  = number of edges, all distinct, no self-loops
    #   seed      = random seed. SAME seed always makes the SAME file,
    #               which is what "reproducible" means in the deliverables
    #   outfile   = where to write it
    #
    # limits from the question paper:  V <= 100000,  E <= 1000000


-- time one graph --

mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np 8 bin/tri_mpi data/generated/Large.txt --time
    # --time = also print the timing breakdown to the screen:
    #     read   = time reading the file (rank 0 only)
    #     bcast  = time sending the edge list to every process
    #     build  = time building the adjacency lists
    #              (every process does this same work - it does NOT
    #               get faster with more processes. this is the limit
    #               on how much speed-up is possible)
    #     count  = time actually counting triangles (this part scales)
    #     reduce = time adding all the process counts together
    #     total  = all of it


-- all 4 process counts, one after another --

for P in 1 2 4 8; do
    echo "--- P=$P ---"
    mpirun --mca pml ^ucx --mca osc ^ucx --bind-to core -np $P bin/tri_mpi data/generated/Large.txt --time
done
    # a loop. P becomes 1, then 2, then 4, then 8.
    # $P inserts the current value into the command.
    # the triangle count must be IDENTICAL for all 4 - if it changes,
    # something is double-counting.
